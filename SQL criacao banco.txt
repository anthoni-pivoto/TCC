
CREATE SEQUENCE tb_lesoes_id_lesao_seq;

                id_lesao BIGINT NOT NULL DEFAULT nextval('tb_lesoes_id_lesao_seq'),

);

                id_exercicio BIGINT NOT NULL DEFAULT nextval('tb_exercicios_id_exercicio_seq'),
                nm_exercicio VARCHAR NOT NULL,
                vid_path VARCHAR,
                CONSTRAINT tb_exercicios_pk PRIMARY KEY (id_exercicio)

                id_lesao BIGINT NOT NULL,

--- 2. Contraindicacao deixa de ser binaria.
---    bloqueio = absoluta: o exercicio sai do catalogo antes de chegar na IA.
---    cautela  = relativa: o exercicio continua disponivel, mas o catalogo
---               enviado ao modelo carrega o aviso explicito.
---    Os vinculos que ja existiam viram 'bloqueio' pelo DEFAULT.

ALTER TABLE tb_exercicio_lesao
    ADD COLUMN nivel VARCHAR NOT NULL DEFAULT 'bloqueio';

ALTER TABLE tb_exercicio_lesao
    ADD CONSTRAINT tb_exercicio_lesao_nivel_ck
    CHECK (nivel IN ('bloqueio', 'cautela'));

COMMENT ON COLUMN tb_exercicio_lesao.nivel IS
    'bloqueio = contraindicacao absoluta (filtro deterministico); cautela = relativa (vai como aviso no prompt)';
